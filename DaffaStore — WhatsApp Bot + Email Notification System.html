// ── services/whatsapp.js ──────────────────────
// WA Notification via Fonnte API (fonnte.com)
// Ganti FONNTE_TOKEN dengan token dari dashboard

const axios = require('axios');

const FONNTE_URL   = 'https://api.fonnte.com/send';
const FONNTE_TOKEN = process.env.FONNTE_TOKEN;

// Kirim pesan WA single
async function sendWA(phone, message) {
  if (!FONNTE_TOKEN) {
    console.warn('[WA] FONNTE_TOKEN belum diset, skip WA');
    return null;
  }
  try {
    const res = await axios.post(FONNTE_URL, {
      target:   phone.replace(/^0/, '62'), // 08xxx → 628xxx
      message:  message,
      countryCode: '62'
    }, {
      headers: { Authorization: FONNTE_TOKEN }
    });
    console.log(\`[WA] Terkirim ke \${phone}\`);
    return res.data;
  } catch (err) {
    console.error('[WA ERROR]', err.message);
    return null;
  }
}

// Kirim notif berdasarkan event
async function notifyOrder(event, order, extra = {}) {
  const data = {
    nama:    extra.name || 'Pelanggan',
    game:    order.game_name,
    item:    \`\${order.amount} \${order.currency}\`,
    userId:  order.user_game_id,
    harga:   (+order.amount_paid).toLocaleString('id-ID'),
    orderId: order.id,
    waktu:   new Date().toLocaleString('id-ID'),
    poin:    Math.floor(order.amount_paid / 1000),
    refCode: extra.refCode || '',
    ...extra
  };

  let msg;
  if (event === 'SUCCESS') msg = msgTopUpSukses(data);
  else if (event === 'PENDING') msg = msgPending(data);
  else if (event === 'FAILED')  msg = msgGagal(data);
  else return;

  return sendWA(extra.phone, msg);
}

module.exports = { sendWA, notifyOrder, msgTopUpSukses, msgPending, msgGagal, msgKomisi };

const nodemailer = require('nodemailer');

const transport = nodemailer.createTransport({
  host: 'smtp.gmail.com', port: 587, secure: false,
  auth: { user: process.env.EMAIL_USER, pass: process.env.EMAIL_PASS }
});

// Kirim email sukses + notif WA sekaligus
async function sendSuccessNotif(order, user) {
  const wa    = require('./whatsapp');
  const tasks = [];

  // Email
  if (order.email) {
    tasks.push(transport.sendMail({
      from:    process.env.EMAIL_FROM,
      to:      order.email,
      subject: \`✅ Top Up Berhasil! \${order.game_name} — \${order.id}\`,
      html:    getEmailTemplate('success', order)
    }));
  }

  // WhatsApp
  if (user?.phone) {
    tasks.push(wa.notifyOrder('SUCCESS', order, {
      name:    user.name,
      phone:   user.phone,
      refCode: user.ref_code
    }));
  }

  // Kirim paralel (tidak saling tunggu)
  await Promise.allSettled(tasks);
}

// Builder HTML email
function getEmailTemplate(type, data) {
  const styles = `font-family:'Segoe UI',sans-serif;max-width:580px;margin:0 auto`;
  const header = (color, icon, title) => \`
    <div style="background:\${color};padding:32px;text-align:center;border-radius:16px 16px 0 0">
      <div style="font-size:3rem;margin-bottom:10px">\${icon}</div>
      <h1 style="color:#fff;margin:0;font-size:1.5rem">\${title}</h1>
    </div>\`;

  if (type === 'success') return \`
    <div style="\${styles}">
      \${header('#10b981', '✅', 'Top Up Berhasil!')}
      <div style="background:#fff;padding:28px;border-radius:0 0 16px 16px">
        <p>Halo <b>\${data.user_name}</b>, top up kamu sudah berhasil diproses!</p>
        <table style="width:100%;border-collapse:collapse;margin:16px 0">
          <tr style="background:#f8f9fa"><td style="padding:10px">Game</td><td style="padding:10px;font-weight:700">\${data.game_name}</td></tr>
          <tr><td style="padding:10px">Item</td><td style="padding:10px;font-weight:700">\${data.amount} \${data.currency}</td></tr>
          <tr style="background:#f8f9fa"><td style="padding:10px">Total</td><td style="padding:10px;font-weight:700;color:#7c3aed">Rp \${(+data.price).toLocaleString()}</td></tr>
          <tr><td style="padding:10px">Order ID</td><td style="padding:10px;font-family:monospace">\${data.id}</td></tr>
        </table>
        <p style="color:#888;font-size:.85rem">Terima kasih telah menggunakan DaffaStore! 🎮</p>
      </div>
    </div>\`;
}

module.exports = { sendSuccessNotif, getEmailTemplate };
